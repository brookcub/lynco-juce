cmake_minimum_required(VERSION 3.22)
project(OboeTimingTests LANGUAGES CXX)
if(NOT ANDROID)
    message(FATAL_ERROR "Use the Android NDK toolchain; this test exercises actual Oboe")
endif()
if(NOT DEFINED OBOE_SOURCE_DIR)
    set(OBOE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../modules/juce_audio_devices/native/oboe")
endif()
add_subdirectory("${OBOE_SOURCE_DIR}" oboe)
add_executable(oboe-timing-tests Main.cpp)
target_compile_features(oboe-timing-tests PRIVATE cxx_std_17)
target_include_directories(oboe-timing-tests PRIVATE "${OBOE_SOURCE_DIR}/src")
target_link_libraries(oboe-timing-tests PRIVATE oboe)
